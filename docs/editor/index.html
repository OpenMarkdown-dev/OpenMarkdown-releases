Put it on a page

<div id="editor"></div>

<script type="module">
  import { createEditorState, mountEditor } from "https://esm.sh/@openmarkdown-dev/editor@0.1.3";

  const editor = mountEditor(
    document.getElementById("editor"),
    createEditorState({ doc: "# Hello\n\nStart typing.\n" }),
  );
  // editor.getDoc() returns the Markdown, byte for byte.
</script>
